---
title: "Get company"
description: "Returns the company that owns the API key."
api: "GET /v1/company"
tag: "Silver"
---

<ResponseField name="id" type="string" required>
  Company UUID.
</ResponseField>

<ResponseField name="name" type="string" required>
  Company name.
</ResponseField>

<ResponseField name="registrationNumber" type="string | null">
  Company registration number.
</ResponseField>

<ResponseField name="email" type="string | null">
  Main contact email.
</ResponseField>

<ResponseField name="phone" type="string | null">
  Main contact phone number.
</ResponseField>

<ResponseField name="address" type="string | null">
  Postal address.
</ResponseField>

<ResponseField name="website" type="string | null">
  Website URL.
</ResponseField>

<ResponseField name="currency" type="string | null">
  ISO 4217 currency code.
</ResponseField>

<ResponseField name="createdAt" type="string" required>
  When the company joined NCZ.
</ResponseField>

<RequestExample>
```bash cURL
curl https://api.nczgroup.com/v1/company \
  -H "x-api-key: YOUR_API_KEY"
```
</RequestExample>
